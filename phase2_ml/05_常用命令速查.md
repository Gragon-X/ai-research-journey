## 一、Git 常用指令

```bash
# 基础
git clone <url>          # 克隆仓库
git status               # 查看状态
git add .                # 添加所有改动
git commit -m "消息"     # 提交
git push                 # 推到 GitHub
git pull                 # 拉最新版本

# 查看
git log --oneline        # 看提交历史
git diff                 # 看改了啥

# 撤销
git reset HEAD <file>    # 撤销 add
git checkout -- <file>   # 撤销文件修改
```

## 二、Linux 常用指令

```bash
# 目录
pwd                      # 当前路径
ls -la                   # 列出文件（详细）
cd <目录>                 # 切换目录
cd ..                    # 上一级
cd ~                     # 家目录
mkdir <文件夹>            # 创建文件夹

# 文件
touch <文件>              # 创建空文件
cp <源> <目标>            # 复制
mv <源> <目标>            # 移动/重命名
rm <文件>                 # 删除（小心！）
rm -r <文件夹>            # 删除文件夹
cat <文件>                # 查看内容

# 搜索
grep "关键词" <文件>      # 搜索文件内容
find . -name "*.py"      # 找文件
```

## 三、NumPy

```python
import numpy as np

# 创建
np.array([1,2,3])            # 从列表创建
np.zeros(10)                 # 全0
np.ones(10)                  # 全1
np.random.seed(42)           # 固定随机种子
np.random.randn(100, 10)     # 正态分布随机数

# 形状
arr.shape                    # 查看形状 (100, 10)
arr.reshape(10, 100)         # 改变形状

# 运算
arr + arr                    # 逐元素加
arr * arr                    # 逐元素乘
arr @ arr.T                  # 矩阵乘法
arr.sum(axis=0)              # 按列求和
arr.mean(axis=0)             # 按列求均值

# 索引
arr[0]                       # 第一行
arr[:, 0]                    # 第一列
arr[arr > 0]                 # 布尔索引

# 工具
np.linalg.norm(arr, axis=1)  # 算范数（KNN里用）
np.argsort(arr)              # 排序后的索引
np.bincount(arr).argmax()    # 投票（KNN里用）
np.round(arr, 3)             # 四舍五入
```

## 四、Pandas

```python
import pandas as pd

# 读取
df = pd.read_csv('文件.csv')  # 读CSV
df.head()                     # 前5行
df.describe()                 # 统计描述
df.info()                     # 列信息

# 选列
df['列名']                    # 选一列
df[['col1', 'col2']]          # 选多列

# 处理
df.dropna()                   # 删缺失值
df.fillna(0)                  # 填缺失值
pd.get_dummies(df, columns=['col'])  # One-Hot编码

# 保存
df.to_csv('新文件.csv', index=False)  # 存CSV
```

## 六、scikit-learn

### 数据
```python
from sklearn.datasets import load_iris, load_wine, load_diabetes
from sklearn.model_selection import train_test_split

data = load_iris()
X, y = data.data, data.target
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42
)
```

### 预处理
```python
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.feature_selection import SelectKBest, f_classif
from sklearn.pipeline import make_pipeline

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

pipe = make_pipeline(StandardScaler(), LogisticRegression())
```

### 分类模型
```python
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import KNeighborsClassifier
from sklearn.tree import DecisionTreeClassifier
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier

model = RandomForestClassifier(n_estimators=100, max_depth=10)
model.fit(X_train, y_train)
y_pred = model.predict(X_test)
```

### 回归模型
```python
from sklearn.linear_model import LinearRegression, Ridge
from sklearn.ensemble import RandomForestRegressor

model = LinearRegression()
model.fit(X_train, y_train)
y_pred = model.predict(X_test)
```

### 评估
```python
from sklearn.metrics import accuracy_score, r2_score, mean_squared_error
from sklearn.model_selection import cross_val_score, GridSearchCV

# 分类
accuracy_score(y_test, y_pred)
cross_val_score(model, X, y, cv=5)

# 回归
r2_score(y_test, y_pred)
cross_val_score(model, X, y, cv=5, scoring='r2')

# 调参
grid = GridSearchCV(model, {'max_depth': [3, 5, None]}, cv=5)
grid.fit(X_train, y_train)
grid.best_params_
```
